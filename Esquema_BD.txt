Table usuarios {
  id uuid [pk]
  correo text [not null]
  nombre text [not null]
  modelo_moto text
  foto_perfil_url text
  apodo text
  color_mapa integer [default: 0]
  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table grupos_ruta {
  id uuid [pk]
  nombre text [not null]
  descripcion text
  codigo_invitacion text [unique, not null]
  creado_por uuid [not null, ref: > usuarios.id]
  activo boolean [default: true]
  foto_url text
  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table miembros_grupo {
  id uuid [pk]
  grupo_id uuid [not null, ref: > grupos_ruta.id]
  usuario_id uuid [not null, ref: > usuarios.id]
  es_admin boolean [default: false]
  fecha_union timestamptz [default: `now()`]
  created_at timestamptz [default: `now()`]
}

Table solicitudes_grupo {
  id uuid [pk]
  grupo_id uuid [not null, ref: > grupos_ruta.id]
  usuario_id uuid [not null, ref: > usuarios.id]
  estado text [default: 'pendiente', note: 'pendiente | aprobado | rechazado']
  fecha_solicitud timestamptz [default: `now()`]
  fecha_resolucion timestamptz
  resuelto_por uuid [ref: > usuarios.id]
  fue_bloqueado_antes boolean [default: false]
  created_at timestamptz [default: `now()`]
}

Table usuarios_bloqueados_grupo {
  id uuid [pk]
  grupo_id uuid [not null, ref: > grupos_ruta.id]
  usuario_id uuid [not null, ref: > usuarios.id]
  bloqueado_por uuid [ref: > usuarios.id]
  motivo text [note: 'expulsado | abandono']
  fecha_bloqueo timestamptz [default: `now()`]
  created_at timestamptz [default: `now()`]
}

Table eventos {
  id uuid [pk]
  titulo text [not null]
  descripcion text [not null]
  fecha_hora timestamptz [not null]
  punto_encuentro text
  creado_por uuid [not null, ref: > usuarios.id]
  max_participants integer
  participants_count integer [default: 0]
  estado text [default: 'activo', note: 'activo | cancelado | finalizado | borrador']
  requires_approval boolean [default: false]
  is_public boolean [default: true]
  foto_url text
  punto_encuentro_lat float
  punto_encuentro_lng float
  destino text
  destino_lat float
  destino_lng float
  reminder_24h_sent boolean [not null, default: false]
  reminder_2h_sent boolean [not null, default: false]
  created_at timestamptz [default: `now()`]
  updated_at timestamptz [default: `now()`]
}

Table evento_grupos {
  evento_id uuid [not null, ref: > eventos.id]
  grupo_id uuid [not null, ref: > grupos_ruta.id]

  indexes {
    (evento_id, grupo_id) [pk]
  }
}

Table participantes_eventos {
  id uuid [pk]
  evento_id uuid [not null, ref: > eventos.id]
  usuario_id uuid [not null, ref: > usuarios.id]
  estado text [default: 'confirmado', note: 'confirmado | posible | no_asiste']
  fecha_registro timestamptz [default: `now()`]
  created_at timestamptz [default: `now()`]
}

Table rutas_realizadas {
    id uuid [pk]
    usuario_id uuid [not null, ref: > usuarios.id]
    nombre_ruta text [not null]
    descripcion_ruta text
    fecha timestamptz [default: `now()`]
    puntos jsonb [not null]
    distancia_km numeric
    duracion_minutos integer
    imagen_url text
    punto_inicio text
    punto_fin text
    tipo_via text
    valor_paisajistico integer [note: '1-5']
    puntos_interes jsonb
    tipo text [not null, default: 'planificada', note: 'planificada: guardada desde el menú de rutas (destino sin navegar) | realizada: trayecto GPS real grabado al finalizar la navegación']                 
    estado text [default: 'activa', note: 'activa | archivada | borrador']
    created_at timestamptz [default: `now()`]                                                               
    updated_at timestamptz [default: `now()`]
  }
